FROM odoo:17.0

COPY ./config/odoo.conf /etc/odoo/

#COPY ./gitman.yaml /usr/lib/python3/dist-packages/odoo/

# El requirements.txt del proyecto viaja en el MISMO COPY, no en uno propio: un
# COPY cuya unica fuente es un glob sin coincidencias lo rechaza el builder
# clasico ("COPY failed: no source files were specified"), y `docker compose
# build` honra DOCKER_BUILDKIT=0 y cae a ese builder. Con una fuente que siempre
# existe al lado, los dos builders lo aceptan con y sin el archivo (#336).
COPY ./install_dependencies.sh requirements.tx[t] /usr/lib/python3/dist-packages/odoo/

USER root

# Dos instalaciones toleran el fallo, por la misma razón: no son requisitos
# del build y su ausencia no debe abortar la imagen.
#
# - pipx no está en los repos de Debian bullseye (odoo:15.0/16.0).
# - libssl-dev/swig/python3-dev/gcc sirven para compilar M2Crypto desde
#   fuente, pero abajo se instala python3-m2crypto, que es el binario ya
#   compilado. Verificado: sin ellos, M2Crypto igual importa.
#
# El sed sobre sources.list corre sólo en bullseye (odoo:15.0/16.0). Esa
# release pasó a oldoldstable y los binarios de debian-security salieron de
# deb.debian.org: el índice sigue anunciando versiones (libssl-dev u8,
# git-man u5) cuyos .deb devuelven 404 en todos los mirrors, y
# archive.debian.org todavía no las tiene. Con el repo presente, apt resuelve
# contra ese índice y falla al descargar, incluso para git, que sí hace falta.
# Sin él, los paquetes salen de bullseye/main, que conserva sus binarios.
# No se pierden parches: hoy no hay de dónde bajarlos.
#
# El rm de EXTERNALLY-MANAGED va acá, en la misma capa y DESPUÉS de los apt
# install, no en una capa previa: python3-dev arrastra el paquete python3.N,
# que vuelve a colocar el archivo. Borrarlo antes no sirve de nada en
# odoo:18.0/19.0 (Ubuntu noble), donde el marcador sí existe.
#
# No usar --break-system-packages para lo mismo: llegó en pip 23 y las imágenes
# viejas no lo soportan (odoo:15.0/16.0 = bullseye, pip 20.3; odoo:17.0 =
# jammy, pip 22.0). Donde el archivo no existe, el rm -f no falla.
RUN . /etc/os-release && \
    if [ "$VERSION_CODENAME" = "bullseye" ]; then \
        sed -i '/security/d' /etc/apt/sources.list; \
    fi && \
    apt update && \
    { apt install -y libssl-dev swig python3-dev gcc || echo "build deps unavailable on this base image, skipping"; } && \
    apt install -y python3-m2crypto && \
    apt install -y git && \
    { apt install -y pipx || echo "pipx no disponible en esta imagen base, se omite"; } && \
    apt clean && \
    rm -rf /var/lib/apt/lists/* && \
    rm -f /usr/lib/python*/EXTERNALLY-MANAGED && \
    python3 -m pip install debugpy gitman

# El requirements.txt del proyecto: lo que sus modulos declaran en
# external_dependencies. install_dependencies.sh no lo cubre, porque los
# `scripts:` de cada fuente de gitman lo corren con `cd` dentro del repo
# clonado y ahi ve el de ese repo (#336).
#
# El COPY va con el glob `requirements.tx[t]` y no con un condicional de Jinja:
# este Dockerfile se renderiza UNA sola vez, en `rkd init`, asi que un
# condicional se evaluaria contra un archivo que todavia no existe. El glob va
# junto a install_dependencies.sh, arriba, por lo que explica ese comentario.
# El de instance si usa condicional: se renderiza en cada deploy.
#
# EXTERNALLY-MANAGED ya se borro arriba, en la capa del apt. --ignore-installed
# porque pip muere con "Cannot uninstall X, RECORD file not found" apenas una
# dependencia pide una version mas nueva de algo que instalo apt; instala sin
# desinstalar, asi que conviven dos versiones -- y por eso mismo actualizar
# cryptography puede dejar roto el pyOpenSSL de apt. El remedio es sumar
# `pyopenssl` al requirements.txt del proyecto, y el import de abajo es lo que
# evita que eso se entregue como un build en verde y un contenedor que no
# arranca: medido, un proyecto que declara authlib sube cryptography y Odoo deja
# de importar.
#
# Se importa `odoo.addons.base` y no `odoo`: desde la 19.0 el paquete `odoo` no
# tiene __init__.py -- es un namespace package (PEP 420) y el arranque se mudo a
# odoo/init.py -- asi que `import odoo` no ejecuta una sola linea de Odoo y el
# chequeo pasa sobre un interprete roto. Medido: con authlib instalado,
# `import odoo` sale 0 en odoo:19.0 y 20.0 mientras el servidor muere con
# "Couldn't load module base". `odoo.addons.base` es lo que importa
# load_server_wide_modules() y sale 1 en las seis series.
RUN if [ -f /usr/lib/python3/dist-packages/odoo/requirements.txt ]; then \
        python3 -m pip install --ignore-installed -r /usr/lib/python3/dist-packages/odoo/requirements.txt && \
        { python3 -c "import odoo.addons.base" || { echo "ERROR: Odoo stopped importing after installing requirements.txt. A dependency upgraded a package Odoo relies on; if the traceback above names OpenSSL/GEN_EMAIL, add 'pyopenssl' to requirements.txt (#336)." >&2; exit 1; }; }; \
    else \
        echo "no requirements.txt in the project, skipping"; \
    fi

CMD python3 -m debugpy --listen 0.0.0.0:8888 /usr/bin/odoo -c /etc/odoo/odoo.conf
